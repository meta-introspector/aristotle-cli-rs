theorem audit_tokens_least_privilege :
    ∀ r ∈ CFTokenPlan.bundle.tokens,
      ∀ p ∈ (CFTokenPlan.bundle.policy r).policies,
        ∀ res ∈ p.resources,
          res.account = CFTokenPlan.account.id ∧
          (∀ pg ∈ p.permissionGroups,
            ∃ c ∈ CFTokenPlan.needs,
              c.group = pg ∧
              (c.token = r ∨ c.token = CFTokenPlan.Role.runtime ∧ r = CFTokenPlan.Role.runtime)) := by
  exact CFTokenPlan.bundle_is_least_privilege

theorem audit_delegation_no_escalation (lvl : Nat) (h : lvl < 3) (p : CFApproval.Proposal) :
    p ∈ CFApproval.menu lvl → p.role ≠ CFApproval.Role.mint := by
  exact CFApproval.mint_no_escalation lvl h p

theorem audit_access_control_conjunction
    (account : AccessControl.Account) (cfg : AccessControl.WorkerCfg)
    (now : Nat) (req : AccessControl.Request) (op : AccessControl.Operation) :
    AccessControl.handle account cfg now req = AccessControl.Response.ok op ↔
      (∃ (ctx : AccessControl.AccessContext) (s : AccessControl.Session),
        req.session = some s ∧
        AccessControl.Deployment.edge account (AccessControl.App.request req) =
          AccessControl.EdgeVerdict.allow ctx ∧
        ctx.aud = cfg.aud ∧
        AccessControl.WorkerCfg.effectiveCaps cfg now ctx req ⊆ AccessControl.capsOf op) := by
  exact AccessControl.handle_ok_iff account cfg now req op
